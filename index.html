quickstart.py Python
from slimme_agent import HermesAgent, Workflow

# Initialise the agent with your API key
agent = HermesAgent(
  api_key="sk-...",
  model="hermes-v2",
)

# Define a workflow in natural language
workflow = Workflow(
  name="invoice-processing",
  trigger="email:inbox",
  steps=[
    "Extract invoice data",
    "Validate against ERP",
    "Route for approval if >€5000",
  ]
)

# Run autonomously
result = agent.run(workflow)
print(result.summary)
# → Processed 847 invoices. Saved 14.3h.
